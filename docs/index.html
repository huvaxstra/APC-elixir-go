Quick Start

Install prerequisites and start Week 1.

bash 
# Install Go
wget https://go.dev/dl/go1.26.4.linux-amd64.tar.gz
sudo tar -C /usr/local -xzf go1.26.4.linux-amd64.tar.gz
echo 'export PATH=$PATH:/usr/local/go/bin' >> ~/.bashrc

# Install Elixir + Erlang (via asdf)
git clone https://github.com/asdf-vm/asdf.git ~/.asdf
asdf plugin add erlang elixir
asdf install erlang 26.2.5
asdf install elixir 1.18.4-otp-27

# Install Docker + kubectl + Terraform
sudo apt install docker.io
curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
curl -fsSL https://releases.hashicorp.com/terraform/1.10.5/terraform_1.10.5_linux_amd64.zip
